// ---------- software WebAuthn authenticator (stands in for iCloud Keychain / GPM) ----------
const subtle = globalThis.crypto.subtle;
const kp = await subtle.generateKey({ name: "ECDSA", namedCurve: "P-256" }, true, ["sign", "verify"]);
const raw = new Uint8Array(await subtle.exportKey("raw", kp.publicKey)); // 0x04||x||y
const x = BigInt(bytesToHex(raw.slice(1, 33)));
const y = BigInt(bytesToHex(raw.slice(33, 65)));
const credId = "spike-credential-1";
const rpId = "wallet.example";
const b64url = (b: Uint8Array) => Buffer.from(b).toString("base64").replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
const derFromRaw = (sig: Uint8Array) => {
  const enc = (v: Uint8Array) => { let i = 0; while (i < v.length - 1 && v[i] === 0) i++; let s = v.slice(i); if (s[0] & 0x80) s = new Uint8Array([0, ...s]); return new Uint8Array([0x02, s.length, ...s]); };
  const r = enc(sig.slice(0, 32)), s = enc(sig.slice(32));
  return new Uint8Array([0x30, r.length + s.length, ...r, ...s]);
};
const getFn = async (opts: any) => {
  const challenge = new Uint8Array(opts.publicKey.challenge);
  const clientDataJSON = `{"type":"webauthn.get","challenge":"${b64url(challenge)}","origin":"https://${rpId}","crossOrigin":false}`;
  const cdj = new TextEncoder().encode(clientDataJSON);
  const authData = concat([sha256(new TextEncoder().encode(rpId)), "0x05", "0x00000001"]); // UP|UV, counter 1
  const msg = concat([authData, sha256(cdj)]);
  const sig = new Uint8Array(await subtle.sign({ name: "ECDSA", hash: "SHA-256" }, kp.privateKey, hexToBytes(msg)));
  return { id: credId, rawId: new TextEncoder().encode(credId).buffer, type: "public-key",
    response: { authenticatorData: hexToBytes(authData).buffer, clientDataJSON: cdj.buffer, signature: derFromRaw(sig).buffer, userHandle: null },
    getClientExtensionResults: () => ({}) };
};
const webAuthnAccount = toWebAuthnAccount({ credential: { id: credId, publicKey: bytesToHex(raw) }, getFn: getFn as any, rpId });

const ONLY = process.env.ONLY?.split(",");
async function sim(label: string, calls: { target: Address; callData: Hex; allowFailure?: boolean }[], stateOverride: any[]) {
  if (ONLY && !ONLY.some((o) => label.startsWith(o))) return null;
  await nap(2500);
  const data = encodeFunctionData({ abi: mcAbi, functionName: "aggregate3", args: [calls.map((c) => ({ allowFailure: c.allowFailure ?? true, ...c }))] });
  try {
    const res = await client.call({ account: RELAYER, to: MULTICALL3, data, stateOverride } as any);
    const out = decodeFunctionResult({ abi: mcAbi, functionName: "aggregate3", data: res.data! }) as any[];
    let gas: bigint | string = "n/a";
    try { gas = await client.estimateGas({ account: RELAYER, to: MULTICALL3, data, stateOverride } as any); } catch (e: any) { gas = "estimate failed: " + (e.shortMessage ?? e.message).slice(0, 80); }
    console.log(`\n[${label}] OK  gas(estimate, whole multicall)=${gas}`);
    out.forEach((r, i) => { const d = r.success ? undefined : decodeRevertData(r.returnData); console.log(`  call ${i}: success=${r.success} ret=${r.returnData.slice(0, 74)}${d ? "  revert=" + d.errorName + " " + d.message : r.success ? "" : "  raw=" + r.returnData.slice(0, 140)}`); });
    return out;
  } catch (e: any) {
    console.log(`\n[${label}] REVERT ${(e.shortMessage ?? e.message).split("\n").slice(0, 3).join(" | ")}`);
    return null;
  }
}
