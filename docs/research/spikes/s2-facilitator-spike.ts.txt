// ==== FILE facilitator.ts ====================================================================
// S2 facilitator: our own x402 `exact` + ERC-7710 facilitator for Arc (eip155:5042). Spike quality, ~150 lines.
// verify(): structural checks on the decoded permissionContext + an eth_call of redeemDelegations FROM the relayer.
// settle(): re-verifies, then either sends the tx (mode "send", anvil/real) or returns the exact tx it would send ("simulate").
import {
  encodeFunctionData, encodePacked, parseAbi, getAddress, isAddressEqual, recoverTypedDataAddress, decodeFunctionResult, zeroAddress,
  BaseError, type Address, type Hex, type PublicClient, type WalletClient,
} from "viem";
import { decodeDelegations, hashDelegation, decodeRevertData } from "@metamask/smart-accounts-kit/utils";
import { ROOT_AUTHORITY, ExecutionMode, type SmartAccountsEnvironment } from "@metamask/smart-accounts-kit";

export type Erc7710Payload = { delegationManager: Address; permissionContext: Hex; delegator: Address };
export type PaymentPayload = { x402Version: 2; accepted: Requirements; payload: Erc7710Payload };
export type Requirements = {
  scheme: "exact"; network: `eip155:${number}`; asset: Address; payTo: Address; amount: string; maxTimeoutSeconds: number;
  extra: { assetTransferMethod: "erc7710"; invoiceHash: Hex; facilitators: Address[] };
};
export type Check = { name: string; ok: boolean; detail?: string };
export type VerifyResult = { isValid: boolean; invalidReason?: string; payer: Address; checks: Check[];
  simulation: { ok: boolean; revert?: string; gas?: bigint } };
export type SimHooks = { stateOverride?: any[]; blockOverrides?: { time?: bigint }; prelude?: { to: Address; data: Hex }[]; multicallCode?: Hex };

const ANY_DELEGATE: Address = "0x0000000000000000000000000000000000000a11";
const GAS_FLOOR = 20_000_000_000n; // Arc: maxFeePerGas below 20 gwei is silently dropped
const dmAbi = parseAbi(["function redeemDelegations(bytes[] _permissionContexts, bytes32[] _modes, bytes[] _executionCallDatas)"]);
const erc20 = parseAbi(["function transfer(address,uint256) returns (bool)"]);
const mcAbi = parseAbi(["struct Call3 { address target; bool allowFailure; bytes callData; }", "struct Result { bool success; bytes returnData; }",
  "function aggregate3(Call3[] calls) payable returns (Result[] returnData)"]);
const dmTypes = { Delegation: [{ name: "delegate", type: "address" }, { name: "delegator", type: "address" }, { name: "authority", type: "bytes32" },
  { name: "caveats", type: "Caveat[]" }, { name: "salt", type: "uint256" }], Caveat: [{ name: "enforcer", type: "address" }, { name: "terms", type: "bytes" }] } as const;

export const buildRedeem = (asset: Address, payTo: Address, amount: bigint, ctx: Hex) => encodeFunctionData({ abi: dmAbi, functionName: "redeemDelegations",
  args: [[ctx], [ExecutionMode.SingleDefault], [encodePacked(["address", "uint256", "bytes"], [asset, 0n, encodeFunctionData({ abi: erc20, functionName: "transfer", args: [payTo, amount] })])]] });

/** Pull a revert reason out of a viem error (string reason, DM custom error or raw data). */
export function revertReason(e: unknown): string {
  let data: Hex | undefined;
  if (e instanceof BaseError) e.walk((x: any) => { if (typeof x?.data === "string" && x.data.startsWith("0x")) data = x.data; else if (typeof x?.data?.data === "string") data = x.data.data; return false; });
  if (data && data !== "0x") return reasonOf(data);
  return ((e as any)?.shortMessage ?? String(e)).split("\n")[0];
}
export function reasonOf(data: Hex): string {
  if (data === "0x") return "empty revert data";
  try { const d = decodeRevertData(data); return d.errorName === "Error" ? String(d.message) : `${d.errorName}${d.message ? " " + d.message : ""}`; }
  catch { return data.slice(0, 74); }
}

export function createFacilitator(cfg: { client: PublicClient; env: SmartAccountsEnvironment; relayer: Address; chainId: number;
  wallet?: WalletClient; requireRedeemer?: boolean }) {
  const allowedManagers = [cfg.env.DelegationManager];
  const E = cfg.env.caveatEnforcers;
  const settled = new Set<string>(); // off-chain replay guard (Arc's IdEnforcer is the real one)

  async function simulate(tx: { to: Address; data: Hex }, sim: SimHooks = {}) {
    const base = { stateOverride: sim.stateOverride, blockOverrides: sim.blockOverrides, gas: 1_500_000n } as any;
    try {
      if (!sim.prelude?.length) { // the real thing: eth_call from the relayer EOA straight to the DelegationManager
        await cfg.client.call({ account: cfg.relayer, ...tx, ...base });
        return { ok: true, gas: await cfg.client.estimateGas({ account: cfg.relayer, ...tx, ...base, gas: undefined }) };
      }
      // simulation of chain history: the relayer address temporarily runs Multicall3 code, so msg.sender is still the relayer
      const calls = [...sim.prelude, tx].map((c) => ({ target: c.to, allowFailure: true, callData: c.data }));
      const res = await cfg.client.call({ to: cfg.relayer, data: encodeFunctionData({ abi: mcAbi, functionName: "aggregate3", args: [calls] }), ...base,
        gas: 8_000_000n, stateOverride: [...(sim.stateOverride ?? []).filter((o: any) => o.address.toLowerCase() !== cfg.relayer.toLowerCase()),
          { ...(sim.stateOverride ?? []).find((o: any) => o.address.toLowerCase() === cfg.relayer.toLowerCase()), address: cfg.relayer, code: sim.multicallCode }] });
      const out = decodeFunctionResult({ abi: mcAbi, functionName: "aggregate3", data: res.data! }) as readonly { success: boolean; returnData: Hex }[];
      const bad = out.slice(0, -1).findIndex((r) => !r.success);
      if (bad >= 0) return { ok: false, revert: `prelude call ${bad} failed: ${reasonOf(out[bad].returnData)}` };
      const last = out[out.length - 1];
      if (last.success) return { ok: true };
      return { ok: false, revert: reasonOf(last.returnData) };
    } catch (e) { return { ok: false, revert: revertReason(e) }; }
  }

  async function verify(p: PaymentPayload, req: Requirements, sim: SimHooks = {}): Promise<VerifyResult> {
    const checks: Check[] = []; const c = (name: string, ok: boolean, detail?: string) => (checks.push({ name, ok, detail }), ok);
    const { delegationManager: dm, permissionContext, delegator } = p.payload;
    c("scheme/network/method", req.scheme === "exact" && req.network === `eip155:${cfg.chainId}` && req.extra.assetTransferMethod === "erc7710");
    c("accepted == requirements (asset, payTo, amount)", isAddressEqual(p.accepted.asset, req.asset) && isAddressEqual(p.accepted.payTo, req.payTo) && p.accepted.amount === req.amount);
    if (!c("delegationManager allowlisted", allowedManagers.some((a) => isAddressEqual(a, dm)), dm))
      return { isValid: false, invalidReason: "unknown_delegation_manager", payer: delegator, checks, simulation: { ok: false, revert: "not simulated" } };
    let chain: any[] = [];
    try { chain = decodeDelegations(permissionContext) as any[]; } catch (e) { c("permissionContext decodes", false, String(e)); }
    if (chain.length) {
      const leaf = chain[0], root = chain[chain.length - 1];
      c("root.delegator == payload.delegator", isAddressEqual(root.delegator, delegator));
      c("root authority is ROOT_AUTHORITY", root.authority.toLowerCase() === ROOT_AUTHORITY.toLowerCase());
      c("chain links (authority = hash(parent), delegator = parent.delegate)", chain.slice(0, -1).every((d, i) =>
        d.authority.toLowerCase() === hashDelegation(chain[i + 1]).toLowerCase() && isAddressEqual(d.delegator, chain[i + 1].delegate)));
      c("leaf delegate is ANY_DELEGATE or a relayer", [ANY_DELEGATE, cfg.relayer].some((a) => isAddressEqual(a, leaf.delegate)));
      const cav = (enf: string) => leaf.caveats.find((x: any) => isAddressEqual(x.enforcer, enf as Address));
      const transfer = encodeFunctionData({ abi: erc20, functionName: "transfer", args: [getAddress(req.payTo), BigInt(req.amount)] });
      const exec = cav(E.ExactExecutionEnforcer)?.terms ?? cav(E.ExactCalldataEnforcer)?.terms;
      c("leaf ExactExecution == USDC.transfer(payTo, amount)", !!exec && [encodePacked(["address", "uint256", "bytes"], [req.asset, 0n, transfer]), transfer]
        .some((t) => t.toLowerCase() === exec.toLowerCase()), exec ? `leaf: ${exec.slice(-104)}` : "missing");
      c("leaf IdEnforcer == invoiceHash", cav(E.IdEnforcer)?.terms?.toLowerCase() === req.extra.invoiceHash.toLowerCase());
      const red = cav(E.RedeemerEnforcer)?.terms as Hex | undefined;
      if (cfg.requireRedeemer || red) c("leaf Redeemer includes relayer", !!red && red.toLowerCase().includes(cfg.relayer.slice(2).toLowerCase()));
      const ts = cav(E.TimestampEnforcer)?.terms as Hex | undefined;
      const now = sim.blockOverrides?.time ?? BigInt(Math.floor(Date.now() / 1000));
      c("leaf Timestamp present and not expired", !!ts && BigInt("0x" + ts.slice(34)) > now, ts ? `before=${BigInt("0x" + ts.slice(34))} now=${now}` : "missing");
      c("invoice not already settled here", !settled.has(req.extra.invoiceHash));
      if (leaf.signature.length === 132) { // EOA leaf (the chip): recover off-chain too
        const who = await recoverTypedDataAddress({ domain: { name: "DelegationManager", version: "1", chainId: cfg.chainId, verifyingContract: dm }, types: dmTypes,
          primaryType: "Delegation", message: { ...leaf, salt: BigInt(leaf.salt) }, signature: leaf.signature }).catch(() => zeroAddress);
        c("leaf signature recovers to leaf.delegator (chip)", isAddressEqual(who, leaf.delegator), who);
      }
    }
    const simulation = await simulate({ to: dm, data: buildRedeem(req.asset, req.payTo, BigInt(req.amount), permissionContext) }, sim);
    const failed = checks.find((x) => !x.ok);
    return { isValid: !failed && simulation.ok, invalidReason: failed ? failed.name : simulation.ok ? undefined : `erc7710_simulation_failed: ${simulation.revert}`, payer: delegator, checks, simulation };
  }

  async function settle(p: PaymentPayload, req: Requirements, mode: "simulate" | "send", sim: SimHooks = {}) {
    const v = await verify(p, req, sim);
    if (!v.isValid) return { success: false as const, errorReason: v.invalidReason, verify: v };
    const tx = { from: cfg.relayer, to: p.payload.delegationManager, data: buildRedeem(req.asset, req.payTo, BigInt(req.amount), p.payload.permissionContext),
      chainId: cfg.chainId, gas: v.simulation.gas ? (v.simulation.gas * 12n) / 10n : 400_000n, maxFeePerGas: GAS_FLOOR * 2n, maxPriorityFeePerGas: 1_000_000_000n, type: "eip1559" as const };
    const costAtFloor = (v.simulation.gas ?? 0n) * GAS_FLOOR; // in wei of native USDC (18 decimals)
    settled.add(req.extra.invoiceHash);
    if (mode === "simulate") return { success: true as const, mode, tx, gasUsedEstimate: v.simulation.gas, costAtFloorWei: costAtFloor, verify: v };
    const hash = await cfg.wallet!.sendTransaction({ ...tx, account: cfg.relayer, chain: null } as any);
    const receipt = await cfg.client.waitForTransactionReceipt({ hash });
    return { success: receipt.status === "success", mode, tx, transaction: hash, gasUsed: receipt.gasUsed, receipt, verify: v };
  }
  return { verify, settle, markSettled: (h: Hex) => settled.add(h), forget: (h: Hex) => settled.delete(h) };
}

// Software WebAuthn authenticator (copied from docs/research/spikes/wallet-accounts-v2-spike.ts.txt): stands in for iCloud Keychain / GPM.
import { bytesToHex, hexToBytes, concat, sha256 } from "viem";
import { toWebAuthnAccount } from "viem/account-abstraction";
const subtle = globalThis.crypto.subtle;
const kp = await subtle.generateKey({ name: "ECDSA", namedCurve: "P-256" }, true, ["sign", "verify"]);
const raw = new Uint8Array(await subtle.exportKey("raw", kp.publicKey)); // 0x04||x||y
const x = BigInt(bytesToHex(raw.slice(1, 33)));
const y = BigInt(bytesToHex(raw.slice(33, 65)));
const credId = "spike-credential-1";
const rpId = "wallet.example";
const b64url = (b: Uint8Array) => Buffer.from(b).toString("base64").replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
const derFromRaw = (sig: Uint8Array) => {
  const enc = (v: Uint8Array) => { let i = 0; while (i < v.length - 1 && v[i] === 0) i++; let s = v.slice(i); if (s[0] & 0x80) s = new Uint8Array([0, ...s]); return new Uint8Array([0x02, s.length, ...s]); };
  const r = enc(sig.slice(0, 32)), s = enc(sig.slice(32));
  return new Uint8Array([0x30, r.length + s.length, ...r, ...s]);
};
const getFn = async (opts: any) => {
  const challenge = new Uint8Array(opts.publicKey.challenge);
  const clientDataJSON = `{"type":"webauthn.get","challenge":"${b64url(challenge)}","origin":"https://${rpId}","crossOrigin":false}`;
  const cdj = new TextEncoder().encode(clientDataJSON);
  const authData = concat([sha256(new TextEncoder().encode(rpId)), "0x05", "0x00000001"]); // UP|UV, counter 1
  const msg = concat([authData, sha256(cdj)]);
  const sig = new Uint8Array(await subtle.sign({ name: "ECDSA", hash: "SHA-256" }, kp.privateKey, hexToBytes(msg)));
  return { id: credId, rawId: new TextEncoder().encode(credId).buffer, type: "public-key",
    response: { authenticatorData: hexToBytes(authData).buffer, clientDataJSON: cdj.buffer, signature: derFromRaw(sig).buffer, userHandle: null },
    getClientExtensionResults: () => ({}) };
};
const webAuthnAccount = toWebAuthnAccount({ credential: { id: credId, publicKey: bytesToHex(raw) }, getFn: getFn as any, rpId });
export { credId, x, y, webAuthnAccount };
