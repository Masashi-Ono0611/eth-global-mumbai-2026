    <script>
      const tabs = [...document.querySelectorAll('[role=tab]')]
      tabs.forEach((t) => t.addEventListener('click', () => {
        tabs.forEach((x) => x.setAttribute('aria-selected', String(x === t)))
        document.getElementById('frame').src = t.dataset.src // reloading the frame ends the other duck's mic and live session
      }))
      const criteria = ['Feels like a conversation', 'Duck character / catchphrases', 'Big QUACK lands', 'Speed (wait after you speak)', 'Safe to demo on stage']
      const box = document.getElementById('score')
      const read = () => { try { return JSON.parse(localStorage.getItem('vita-compare') || '{}') } catch { return {} } }
      const save = (v) => { try { localStorage.setItem('vita-compare', JSON.stringify(v)) } catch {} }
      const state = read()
      const pick = (key) => { const s = document.createElement('select'); s.add(new Option('–', '')); for (let i = 1; i <= 5; i++) s.add(new Option(String(i), String(i))); s.value = state[key] ?? ''; s.addEventListener('change', () => { state[key] = s.value; save(state); verdict() }); return s }
      for (const c of criteria) {
        const l = document.createElement('label'); l.append(c)
        const a = pick(`A:${c}`); a.setAttribute('aria-label', `${c}, A`); const b = pick(`B:${c}`); b.setAttribute('aria-label', `${c}, B`)
        l.append(a, b); box.append(l)
      }
      const head = document.createElement('label'); head.innerHTML = '<span></span><b style="color:var(--duck)">A</b><b style="color:var(--satchel)">B</b>'; box.insertBefore(head, box.children[1])
      const verdict = () => {
        const sum = (p) => criteria.reduce((n, c) => n + (Number(state[`${p}:${c}`]) || 0), 0)
        const a = sum('A'), b = sum('B')
        document.getElementById('verdict').textContent = a || b ? `A ${a} : B ${b}${a === b ? ' (tie)' : a > b ? ' → A (0G) wins' : ' → B (GPT-Live) wins'}` : ''
      }
      verdict()
    </script>
