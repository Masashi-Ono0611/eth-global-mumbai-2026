    <script>
      const $ = (id) => document.getElementById(id)
      const setState = (s) => (document.body.dataset.state = s)
      const history = []
      let ctx, analyser, player, playerSource

      const audio = () => {
        if (!ctx) {
          ctx = new AudioContext()
          analyser = ctx.createAnalyser(); analyser.fftSize = 512
          player = new Audio(); player.preservesPitch = false; player.webkitPreservesPitch = false
          playerSource = ctx.createMediaElementSource(player)
          playerSource.connect(analyser); analyser.connect(ctx.destination)
        }
        if (ctx.state === 'suspended') ctx.resume()
        return ctx
      }

      // A synthesized quack (no sample files): a falling buzzy tone through a nasal band-pass.
      const quack = (when = 0, pitch = 1) => {
        if (!$('quackOn').checked) return
        const c = audio(); const t = c.currentTime + when
        const osc = c.createOscillator(); osc.type = 'sawtooth'
        osc.frequency.setValueAtTime(620 * pitch, t); osc.frequency.exponentialRampToValueAtTime(380 * pitch, t + 0.17)
        const osc2 = c.createOscillator(); osc2.type = 'square'
        osc2.frequency.setValueAtTime(310 * pitch, t); osc2.frequency.exponentialRampToValueAtTime(200 * pitch, t + 0.17)
        const band = c.createBiquadFilter(); band.type = 'bandpass'; band.frequency.value = 1350; band.Q.value = 4
        const nasal = c.createBiquadFilter(); nasal.type = 'peaking'; nasal.frequency.value = 2600; nasal.gain.value = 9
        const gain = c.createGain()
        gain.gain.setValueAtTime(0.0001, t); gain.gain.exponentialRampToValueAtTime(0.5, t + 0.015); gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.21)
        osc.connect(band); osc2.connect(band); band.connect(nasal); nasal.connect(gain); gain.connect(c.destination)
        osc.start(t); osc2.start(t); osc.stop(t + 0.23); osc2.stop(t + 0.23)
      }

      // Beak follows the voice's loudness.
      const level = new Uint8Array(512)
      const mouth = () => {
        let open = 0
        if (analyser && !player.paused) {
          analyser.getByteTimeDomainData(level)
          let sum = 0; for (const v of level) sum += (v - 128) ** 2
          open = Math.min(1, Math.sqrt(sum / level.length) / 28)
        }
        $('jaw').style.transform = `rotate(${open * 16}deg) translateY(${open * 6}px)`
        requestAnimationFrame(mouth)
      }
      requestAnimationFrame(mouth)

      // Recording: raw PCM, sent as a 16 kHz mono WAV (0G's omni models take WAV as a data URI).
      // `holding` is the button state; recording starts only if it is still held once the mic permission resolves.
      let stream, source, processor, chunks = [], recording = false, holding = false, startedAt = 0
      const startRec = async () => {
        if (recording || holding || $('talk').disabled) return
        holding = true
        const c = audio()
        try { stream ??= await navigator.mediaDevices.getUserMedia({ audio: { channelCount: 1, echoCancellation: true, noiseSuppression: true } }) }
        catch { holding = false; $('status').textContent = 'Microphone blocked. You can type instead.'; return }
        if (!holding) { $('status').textContent = 'Microphone ready. Hold the button while you speak.'; return }
        chunks = []; recording = true; startedAt = performance.now()
        source = c.createMediaStreamSource(stream)
        processor = c.createScriptProcessor(4096, 1, 1)
        processor.onaudioprocess = (e) => { if (recording) chunks.push(new Float32Array(e.inputBuffer.getChannelData(0))) }
        source.connect(processor); processor.connect(c.destination)
        $('talk').setAttribute('aria-pressed', 'true'); $('talk').textContent = 'Listening…'; setState('listening'); $('status').textContent = ''
      }
      const stopRec = () => {
        holding = false
        if (!recording) return
        recording = false
        source.disconnect(); processor.disconnect()
        $('talk').setAttribute('aria-pressed', 'false'); $('talk').textContent = 'Hold to talk'
        if (performance.now() - startedAt < 400) { setState('idle'); $('status').textContent = 'Hold the button while you speak.'; return }
        send({ audioB64: toWavBase64(chunks, ctx.sampleRate) })
      }
      const toWavBase64 = (parts, rate) => {
        const total = parts.reduce((n, p) => n + p.length, 0)
        const ratio = rate / 16000, length = Math.floor(total / ratio)
        const flat = new Float32Array(total); let o = 0; for (const p of parts) { flat.set(p, o); o += p.length }
        const buf = new DataView(new ArrayBuffer(44 + length * 2))
        const str = (at, s) => { for (let i = 0; i < s.length; i++) buf.setUint8(at + i, s.charCodeAt(i)) }
        str(0, 'RIFF'); buf.setUint32(4, 36 + length * 2, true); str(8, 'WAVE'); str(12, 'fmt ')
        buf.setUint32(16, 16, true); buf.setUint16(20, 1, true); buf.setUint16(22, 1, true); buf.setUint32(24, 16000, true)
        buf.setUint32(28, 32000, true); buf.setUint16(32, 2, true); buf.setUint16(34, 16, true); str(36, 'data'); buf.setUint32(40, length * 2, true)
        for (let i = 0; i < length; i++) {
          const s = Math.max(-1, Math.min(1, flat[Math.floor(i * ratio)]))
          buf.setInt16(44 + i * 2, s < 0 ? s * 0x8000 : s * 0x7fff, true)
        }
        let bin = ''; const bytes = new Uint8Array(buf.buffer)
        for (let i = 0; i < bytes.length; i += 0x8000) bin += String.fromCharCode(...bytes.subarray(i, i + 0x8000))
        return btoa(bin)
      }

      // Catchphrases in duck yellow, puns in satchel teal; built as DOM nodes (the reply is model text, never HTML).
      const CATCH = /(quack quack!|what the duck\?!|waddle waddle!|qua+ck!|budget first, snacks second, quack!|that's a boss decision, quack!)/gi
      const PUN = /\b(egg-cellent|quack-tually|un-beak-lievable|in-duck-tive|no fowl play|duck-ument|feather-weight|quackers|a bill for my bill)\b/gi
      const showReply = (text) => {
        const el = $('reply'); el.replaceChildren()
        const marks = []
        for (const re of [CATCH, PUN]) for (const m of text.matchAll(re)) marks.push({ at: m.index, end: m.index + m[0].length, cls: re === CATCH ? 'catch' : 'pun' })
        marks.sort((a, b) => a.at - b.at)
        let pos = 0
        for (const m of marks) {
          if (m.at < pos) continue
          el.append(text.slice(pos, m.at))
          const span = document.createElement('span'); span.className = m.cls; span.textContent = text.slice(m.at, m.end); el.append(span)
          pos = m.end
        }
        el.append(text.slice(pos))
        // The final "QUACK!" gets its own span so it alone can grow while the slow QUACK plays.
        const closer = [...el.querySelectorAll('.catch')].pop()
        const m = closer && /^(.*?)(QUACK!)$/i.exec(closer.textContent)
        if (m) {
          closer.textContent = m[1]
          const word = document.createElement('span'); word.className = 'catch quackword'; word.textContent = m[2]
          closer.after(word)
        }
      }
      // While the big QUACK plays, the closing "QUACK!" in the reply grows.
      const bigQuack = (on) => {
        document.querySelector('#reply .quackword')?.classList.toggle('big', on)
      }
      const hop = () => { document.body.classList.remove('hop'); void document.body.offsetWidth; document.body.classList.add('hop'); setTimeout(() => document.body.classList.remove('hop'), 1200) }

      // The pitch-up also speeds playback up, so 0G speaks slower by the same factor: heard tempo = Talk speed.
      const ttsSpeed = () => Math.min(1.5, Math.max(0.5, Number($('tempo').value) / Number($('pitch').value)))
      const send = async (payload) => {
        $('talk').disabled = true; setState('thinking'); $('status').textContent = 'Vita is thinking on 0G…'
        quack(0, 1.1)
        try {
          const r = await fetch('/talk', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ ...payload, history, voice: $('voice').value, speed: ttsSpeed() }) })
          const j = await r.json()
          if (!r.ok) throw new Error(j.error || r.status)
          history.push({ heard: j.heard || payload.text || '', reply: j.reply })
          $('said').textContent = j.heard || payload.text ? `You: ${j.heard || payload.text}` : ''
          showReply(j.reply)
          $('timing').textContent = `think ${(j.ms / 1000).toFixed(1)} s, voice ${(j.speakMs / 1000).toFixed(1)} s`
          $('status').textContent = ''
          setState('speaking')
          hop(); quack(0, 1); quack(0.24, 1.05)
          player.src = 'data:audio/mpeg;base64,' + j.mp3B64
          player.playbackRate = Number($('pitch').value)
          await new Promise((ok) => setTimeout(ok, 480))
          await player.play()
          await new Promise((ok) => (player.onended = ok))
          // The closer: a beat of silence, then a slow, big QUAAACK with a hop, the word lit up and quacks on top.
          await new Promise((ok) => setTimeout(ok, 280))
          bigQuack(true)
          hop(); quack(0, 1.15)
          if (j.quackB64) {
            player.src = 'data:audio/mpeg;base64,' + j.quackB64
            player.playbackRate = Number($('pitch').value)
            await player.play()
            await new Promise((ok) => (player.onended = ok))
          }
          quack(0, 1.05); quack(0.18, 1.25)
          await new Promise((ok) => setTimeout(ok, 500))
          bigQuack(false)
        } catch (e) {
          $('status').textContent = `Quack… 0G did not answer (${String(e.message || e).slice(0, 80)}). Try again.`
        } finally {
          $('talk').disabled = false; setState('idle')
        }
      }

      const talk = $('talk')
      talk.addEventListener('pointerdown', (e) => { e.preventDefault(); talk.setPointerCapture(e.pointerId); startRec() })
      talk.addEventListener('pointerup', stopRec)
      talk.addEventListener('pointercancel', stopRec)
      addEventListener('keydown', (e) => { if (e.code === 'Space' && !e.repeat && !e.target.closest('input, select, summary, form button')) { e.preventDefault(); startRec() } })
      addEventListener('keyup', (e) => { if (e.code === 'Space' && (recording || holding)) { e.preventDefault(); stopRec() } })
      $('typeForm').addEventListener('submit', (e) => {
        e.preventDefault(); const text = $('typeInput').value.trim(); if (!text || talk.disabled) return
        $('typeInput').value = ''; audio(); send({ text })
      })

      fetch('/voices').then((r) => r.json()).then((v) => {
        for (const name of v.voices) $('voice').add(new Option(name.replace(/_/g, ' '), name))
        $('chatModel').textContent = v.chatModel; $('ttsModel').textContent = v.ttsModel
      })
    </script>
