        <div class="bubble" aria-live="polite">
          <p class="said" id="said"></p>
          <p class="reply" id="reply"><span class="catch">Quack quack!</span> Press Start and just talk to me.</p>

      <div class="controls">
        <div class="row">
          <button class="big" id="start">Start live</button>
          <button class="quiet" id="mute" aria-pressed="false" disabled>Mute mic</button>
          <button class="quiet" id="end" disabled>End</button>
        </div>
        <form class="type" id="typeForm" autocomplete="off">
          <input id="typeInput" placeholder="…or type to Vita (while live)" aria-label="Type a message to Vita" />
          <button type="submit">Send</button>
        </form>
        <p class="credit">Hearing, thinking and voice: <b id="model">GPT-Live</b> (OpenAI) · <span id="clock"></span></p>

    <script>
      const $ = (id) => document.getElementById(id)
      const setState = (s) => (document.body.dataset.state = s)
      let pc, dc, mic, ctx, analyser, timer, endsAt

      // Same highlighting as the 0G page: catchphrases yellow, puns teal (DOM nodes, never HTML).
      const CATCH = /(quack quack!|what the duck\?!|waddle waddle!|qua+ck!|budget first, snacks second, qua+ck!|that's a boss decision, qua+ck!)/gi
      const PUN = /\b(egg-cellent|quack-tually|un-beak-lievable|in-duck-tive|no fowl play|duck-ument|feather-weight|quackers|a bill for my bill)\b/gi
      const showReply = (text) => {
        const el = $('reply'); el.replaceChildren()
        const marks = []
        for (const re of [CATCH, PUN]) for (const m of text.matchAll(re)) marks.push({ at: m.index, end: m.index + m[0].length, cls: re === CATCH ? 'catch' : 'pun' })
        marks.sort((a, b) => a.at - b.at)
        let pos = 0
        for (const m of marks) {
          if (m.at < pos) continue
          el.append(text.slice(pos, m.at))
          const s = document.createElement('span'); s.className = m.cls; s.textContent = text.slice(m.at, m.end); el.append(s)
          pos = m.end
        }
        el.append(text.slice(pos))
      }
      const hop = () => { document.body.classList.remove('hop'); void document.body.offsetWidth; document.body.classList.add('hop'); setTimeout(() => document.body.classList.remove('hop'), 1200) }

      // Beak follows the live voice.
      const level = new Uint8Array(512)
      const mouth = () => {
        let open = 0
        if (analyser) {
          analyser.getByteTimeDomainData(level)
          let sum = 0; for (const v of level) sum += (v - 128) ** 2
          open = Math.min(1, Math.sqrt(sum / level.length) / 28)
        }
        $('jaw').style.transform = `rotate(${open * 16}deg) translateY(${open * 6}px)`
        requestAnimationFrame(mouth)
      }
      requestAnimationFrame(mouth)

      // Transcripts arrive in small pieces: a new user line starts a new turn.
      let said = '', reply = '', closed = false
      const onEvent = (raw) => {
        let ev; try { ev = JSON.parse(raw) } catch { return }
        if (ev.type === 'session.started') { $('status').textContent = 'Live: just talk'; setState('listening') }
        else if (ev.type === 'session.closed') stop('Session ended')
        else if (ev.type === 'error') $('status').textContent = `Error: ${JSON.stringify(ev.error ?? ev).slice(0, 120)}`
        else if (ev.type === 'session.input_transcript.delta' && typeof ev.delta === 'string') {
          if (reply) { said = ''; reply = ''; closed = false }
          said += ev.delta; $('said').textContent = `You: ${said}`
        } else if (ev.type === 'session.output_transcript.delta' && typeof ev.delta === 'string') {
          // No synthesized quacks here: the live voice does its own QUAAACK. The duck still hops.
          if (!reply) hop()
          reply += ev.delta; showReply(reply.replace(/\[[^\]]*\]\s*/g, '').trim()) // drop voice cues such as [breath]
          if (!closed && /(second|decision), qua+ck!?\s*$/i.test(reply)) { closed = true; setTimeout(hop, 900) }
        }
      }

      const send = (ev) => { if (dc?.readyState === 'open') dc.send(JSON.stringify(ev)) }
      const start = async () => {
        if (pc) return
        $('start').disabled = true; $('status').textContent = 'Connecting…'
        try {
          ctx = new AudioContext(); analyser = ctx.createAnalyser(); analyser.fftSize = 512
          pc = new RTCPeerConnection()
          const out = new Audio(); out.autoplay = true
          pc.ontrack = (e) => { out.srcObject = e.streams[0]; ctx.createMediaStreamSource(e.streams[0]).connect(analyser) }
          mic = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true } })
          mic.getTracks().forEach((t) => pc.addTrack(t, mic))
          dc = pc.createDataChannel('oai-events'); dc.onmessage = (m) => onEvent(m.data)
          await pc.setLocalDescription(await pc.createOffer())
          const r = await fetch('/live/session', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ sdp: pc.localDescription.sdp }) })
          const j = await r.json()
          if (!j.ok) throw new Error(j.error || r.status)
          await pc.setRemoteDescription({ type: 'answer', sdp: j.transport.sdp })
          $('model').textContent = `${j.model}, voice ${j.voice}`
          endsAt = Date.now() + j.maxSessionSeconds * 1000
          timer = setInterval(() => { const s = Math.max(0, Math.round((endsAt - Date.now()) / 1000)); $('clock').textContent = `${s} s left`; if (!s) stop('Time is up') }, 500)
          $('mute').disabled = false; $('end').disabled = false
        } catch (e) { stop(`Could not start: ${String(e.message || e).slice(0, 120)}`) }
      }
      const stop = (why) => {
        try { send({ type: 'session.close' }) } catch {}
        mic?.getTracks().forEach((t) => t.stop()); pc?.close(); ctx?.close()
        pc = dc = mic = ctx = analyser = null; clearInterval(timer); $('clock').textContent = ''
        $('start').disabled = false; $('mute').disabled = true; $('end').disabled = true; $('mute').setAttribute('aria-pressed', 'false'); $('mute').textContent = 'Mute mic'
        $('status').textContent = why || 'Not connected'; setState('idle')
      }
      $('start').addEventListener('click', start)
      $('end').addEventListener('click', () => stop('Ended'))
      $('mute').addEventListener('click', () => {
        const muted = $('mute').getAttribute('aria-pressed') !== 'true'
        mic?.getAudioTracks().forEach((t) => (t.enabled = !muted))
        $('mute').setAttribute('aria-pressed', String(muted)); $('mute').textContent = muted ? 'Unmute mic' : 'Mute mic'
        $('status').textContent = muted ? 'Mic muted (still connected)' : 'Live: just talk'
      })
      $('typeForm').addEventListener('submit', (e) => {
        e.preventDefault(); const text = $('typeInput').value.trim(); if (!text) return
        if (dc?.readyState !== 'open') { $('status').textContent = 'Not live yet: press Start and wait for "Live".'; return }
        $('typeInput').value = ''; said = text; reply = ''; closed = false; $('said').textContent = `You: ${text}`
        send({ type: 'response.item.create', item: { type: 'message', role: 'user', content: [{ type: 'input_text', text }] } })
        send({ type: 'response.create' })
      })
    </script>
