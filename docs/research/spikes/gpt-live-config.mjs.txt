const NO_ARGS = { type: 'object', additionalProperties: false, properties: {}, required: [] };

export const readLiveEnv = (env) => {
  const num = (key, fallback, min, max) => {
    const raw = env[key];
    if (raw === undefined || raw === '') return fallback;
    const n = Number(raw);
    return !Number.isFinite(n) || n < min || n > max ? fallback : Math.trunc(n);
  };
  return {
    enabled: env.LIVE_ENABLED !== 'false',
    apiKey: env.OPENAI_API_KEY || undefined,
    model: env.LIVE_MODEL || 'gpt-live-1',
    voice: (env.LIVE_VOICE || 'gleam').toLowerCase(),
    backendModel: env.LIVE_BACKEND_MODEL || 'gpt-5.6-terra',
    backendReasoning: env.LIVE_BACKEND_REASONING || 'low',
    maxSessionSeconds: num('LIVE_MAX_SESSION_SECONDS', 180, 1, 900),
    ipHourly: num('LIVE_IP_HOURLY_LIMIT', 6, 0, 1000),
    globalDaily: num('LIVE_GLOBAL_DAILY_LIMIT', 40, 0, 10000),
  };
};

/** サーバーが固定するセッション設定。ブラウザは変更できない(allowed_client_events で 3 種類に限る)。 */
export const buildLiveConfig = (env) => ({
  model: env.model,
  instructions: DUCK_INSTRUCTIONS,
  audio: { output: { voice: env.voice } },
  client: { data_channel: { allowed_client_events: [...LIVE_ALLOWED_CLIENT_EVENTS] } },
  delegation: {
    type: 'responses',
    responses: {
      model: env.backendModel, instructions: BACKEND_INSTRUCTIONS,
      tools: [GO_TO_TOOL, STOP_TOOL, GET_STATUS_TOOL],
      tool_choice: 'auto', parallel_tool_calls: false, reasoning: { effort: env.backendReasoning }, max_output_tokens: 400,
    },
  },
});
