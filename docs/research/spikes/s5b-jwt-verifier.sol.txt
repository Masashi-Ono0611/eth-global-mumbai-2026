pragma solidity ^0.8.24;

import {RSA} from "@openzeppelin/contracts/utils/cryptography/RSA.sol";
import {Base64} from "@openzeppelin/contracts/utils/Base64.sol";
import {Strings} from "@openzeppelin/contracts/utils/Strings.sol";

/// @title HumanApprovalVerifier (spike S5b)
/// @notice Verifies a World IdP (Agent Pass) RS256 ID token on-chain and binds it to one payment.
///         Prototype of the HumanApprovalEnforcer core. NOT AUDITED.
/// @dev JSON handling is a deliberately minimal, fail-closed scanner of the TOP-LEVEL object only:
///      - the payload must be exactly one JSON object (optional surrounding whitespace);
///      - top-level keys are compared byte-for-byte; any top-level key containing a backslash reverts
///        (so `"nonce"` cannot alias `nonce`);
///      - nested objects/arrays and other string values are skipped with full string/escape tracking,
///        so text that *looks like* `"nonce":"..."` inside a value or a nested object is never a match;
///      - each claim we read must appear exactly once (duplicates revert: parsers disagree on them);
///      - string claims we read must contain no backslash (no escape decoding is done, fail closed);
///      - numeric claims must be plain unsigned integers (no sign, fraction or exponent), <= 15 digits;
///      - `aud` must be a single string (World issues a string; an array reverts);
///      - `amr` must be an array of strings.
contract HumanApprovalVerifier {
    using Strings for uint256;

    // ---------------------------------------------------------------- errors
    error NotAdmin();
    error BadModulus();
    error MalformedJwt();
    error UnsupportedAlg();
    error CritHeader();
    error UnknownKid();
    error BadSignature();
    error MalformedJson(uint256 pos);
    error EscapedKey(uint256 pos);
    error EscapedClaim(bytes32 key);
    error DuplicateClaim(bytes32 key);
    error MissingClaim(bytes32 key);
    error WrongType(bytes32 key);
    error BadIssuer();
    error BadAudience();
    error BadSubject();
    error BadNonce();
    error Expired();
    error AuthTooOld();
    error AuthInFuture();
    error IssuedInFuture();
    error NoProofOfPersonhood();
    error PasskeyOnly();
    error BadAcr();
    error JtiUsed();
    error NonceUsed();

    event KeyRegistered(string kid, bytes32 kidHash);
    event KeyRevoked(bytes32 kidHash);
    event ApprovalConsumed(bytes32 indexed nonce, bytes32 indexed subHash, bytes32 jtiHash, uint256 authTime);

    // ------------------------------------------------------------- constants
    uint8 private constant T_STRING = 1;
    uint8 private constant T_NUMBER = 2;
    uint8 private constant T_ARRAY = 3;
    uint8 private constant T_OTHER = 4;

    bytes32 private constant K_ALG = keccak256("alg");
    bytes32 private constant K_KID = keccak256("kid");
    bytes32 private constant K_CRIT = keccak256("crit");
    bytes32 private constant K_ISS = keccak256("iss");
    bytes32 private constant K_SUB = keccak256("sub");
    bytes32 private constant K_AUD = keccak256("aud");
    bytes32 private constant K_EXP = keccak256("exp");
    bytes32 private constant K_IAT = keccak256("iat");
    bytes32 private constant K_AUTH = keccak256("auth_time");
    bytes32 private constant K_NONCE = keccak256("nonce");
    bytes32 private constant K_JTI = keccak256("jti");
    bytes32 private constant K_AMR = keccak256("amr");
    bytes32 private constant K_ACR = keccak256("acr");

    bytes32 private constant RS256 = keccak256("RS256");
    bytes32 private constant POP = keccak256("pop");
    bytes32 private constant MFA = keccak256("mfa");
    bytes private constant E65537 = hex"010001";
    uint256 private constant CLOCK_SKEW = 60;

    // ----------------------------------------------------------------- state
    address public admin;
    bytes32 public immutable issuerHash; // keccak("https://auth.world.org")
    uint256 public immutable maxAuthAge; // seconds between World proof (auth_time) and use
    mapping(bytes32 kidHash => bytes modulus) public modulusOf;
    mapping(bytes32 acrHash => bool) public acrAllowed;
    mapping(bytes32 jtiHash => bool) public jtiUsed;
    mapping(bytes32 nonce => bool) public nonceUsed;

    struct Span {
        uint256 start; // first byte of the value (inside the quotes for strings)
        uint256 end; // exclusive
        uint8 kind; // 0 = absent
        bool esc; // string value contained a backslash escape
    }

    struct Claims {
        bytes32 subHash;
        bytes32 jtiHash;
        uint256 exp;
        uint256 iat;
        uint256 authTime;
    }

    constructor(string memory issuer, uint256 maxAuthAge_, string[] memory acrs) {
        admin = msg.sender;
        issuerHash = keccak256(bytes(issuer));
        maxAuthAge = maxAuthAge_;
        for (uint256 i; i < acrs.length; ++i) acrAllowed[keccak256(bytes(acrs[i]))] = true;
    }

    // ----------------------------------------------------------------- admin
    modifier onlyAdmin() {
        if (msg.sender != admin) revert NotAdmin();
        _;
    }

    /// @notice Registers an RSA public key (e = 65537). The kid is DERIVED as the RFC 7638 JWK thumbprint,
    ///         which is how World IdP's live JWKS names its keys, so the admin cannot mislabel a key.
    function registerKey(bytes calldata n) external onlyAdmin returns (string memory kid) {
        if (n.length < 256 || n.length > 512 || n[0] == 0) revert BadModulus();
        bytes memory jwk = abi.encodePacked('{"e":"AQAB","kty":"RSA","n":"', Base64.encodeURL(n), '"}');
        kid = Base64.encodeURL(abi.encodePacked(sha256(jwk)));
        bytes32 h = keccak256(bytes(kid));
        modulusOf[h] = n;
        emit KeyRegistered(kid, h);
    }

    function revokeKey(string calldata kid) external onlyAdmin {
        bytes32 h = keccak256(bytes(kid));
        delete modulusOf[h];
        emit KeyRevoked(h);
    }

    function setAcr(string calldata acr, bool allowed) external onlyAdmin {
        acrAllowed[keccak256(bytes(acr))] = allowed;
    }

    // ------------------------------------------------------------ main entry
    /// @notice Verify `jwt` and consume its jti and nonce. Reverts on any failure.
    /// @param nonce the EIP-712 hash of the invoice; the token's `nonce` claim must be its 0x-lowercase-hex form.
    function verifyAndConsume(bytes calldata jwt, bytes32 nonce, string calldata sub, string calldata aud)
        external
        returns (Claims memory c)
    {
        c = verify(jwt, nonce, sub, aud);
        if (jtiUsed[c.jtiHash]) revert JtiUsed();
        if (nonceUsed[nonce]) revert NonceUsed();
        jtiUsed[c.jtiHash] = true;
        nonceUsed[nonce] = true;
        emit ApprovalConsumed(nonce, c.subHash, c.jtiHash, c.authTime);
    }

    /// @notice Stateless verification (signature + claims), no replay bookkeeping.
    function verify(bytes calldata jwt, bytes32 nonce, string calldata sub, string calldata aud)
        public
        view
        returns (Claims memory c)
    {
        (uint256 d1, uint256 d2) = _dots(jwt);

        // ---- header: alg == RS256, kid -> registered modulus, no `crit`
        bytes memory n;
        {
            bytes memory h = Base64.decode(string(jwt[:d1]));
            bytes32[] memory hk = new bytes32[](3);
            hk[0] = K_ALG;
            hk[1] = K_KID;
            hk[2] = K_CRIT;
            Span[] memory hs = _scan(h, hk);
            if (hs[2].kind != 0) revert CritHeader();
            if (_strHash(h, hs[0], K_ALG) != RS256) revert UnsupportedAlg();
            n = modulusOf[_strHash(h, hs[1], K_KID)];
            if (n.length == 0) revert UnknownKid();
        }

        // ---- signature over the ASCII bytes of `header.payload`
        {
            bytes memory s = Base64.decode(string(jwt[d2 + 1:]));
            if (!RSA.pkcs1Sha256(sha256(jwt[:d2]), s, E65537, n)) revert BadSignature();
        }

        // ---- claims
        bytes memory p = Base64.decode(string(jwt[d1 + 1:d2]));
        bytes32[] memory keys = new bytes32[](10);
        keys[0] = K_ISS;
        keys[1] = K_SUB;
        keys[2] = K_AUD;
        keys[3] = K_EXP;
        keys[4] = K_IAT;
        keys[5] = K_AUTH;
        keys[6] = K_NONCE;
        keys[7] = K_JTI;
        keys[8] = K_AMR;
        keys[9] = K_ACR;
        Span[] memory v = _scan(p, keys);

        if (_strHash(p, v[0], K_ISS) != issuerHash) revert BadIssuer();
        c.subHash = _strHash(p, v[1], K_SUB);
        if (c.subHash != keccak256(bytes(sub))) revert BadSubject();
        if (_strHash(p, v[2], K_AUD) != keccak256(bytes(aud))) revert BadAudience();
        if (_strHash(p, v[6], K_NONCE) != keccak256(bytes(uint256(nonce).toHexString(32)))) revert BadNonce();
        c.jtiHash = _strHash(p, v[7], K_JTI);
        if (!acrAllowed[_strHash(p, v[9], K_ACR)]) revert BadAcr();

        c.exp = _uint(p, v[3], K_EXP);
        c.iat = _uint(p, v[4], K_IAT);
        c.authTime = _uint(p, v[5], K_AUTH);
        if (block.timestamp >= c.exp) revert Expired();
        if (c.iat > block.timestamp + CLOCK_SKEW) revert IssuedInFuture();
        if (c.authTime > block.timestamp + CLOCK_SKEW) revert AuthInFuture();
        if (c.authTime + maxAuthAge < block.timestamp) revert AuthTooOld();

        _checkAmr(p, v[8]);
    }

    // ------------------------------------------------------------- internals
    // Byte loops are in assembly: the checked Solidity versions cost ~140-450 gas/byte (spike measurement).

    uint256 private constant ONES = 0x0101010101010101010101010101010101010101010101010101010101010101;
    uint256 private constant LOW7 = 0x7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f7f;

    /// @dev SWAR helpers. `_zeroMask(x)` sets the high bit of every byte of `x` that is 0x00 (exact, no
    ///      cross-byte carries). `_firstFlag(z)` returns the index (0 = lowest address) of the first flagged
    ///      byte via a 5-step binary search (no CLZ opcode needed). `_tailMask(n)` keeps the first n bytes.
    function _zeroMask(uint256 x) internal pure returns (uint256 z) {
        assembly ("memory-safe") {
            z := not(or(or(add(and(x, LOW7), LOW7), x), LOW7))
        }
    }

    function _firstFlag(uint256 z) internal pure returns (uint256 idx) {
        assembly ("memory-safe") {
            let r := 0
            if gt(z, 0xffffffffffffffffffffffffffffffff) { z := shr(128, z) r := 128 }
            if gt(z, 0xffffffffffffffff) { z := shr(64, z) r := add(r, 64) }
            if gt(z, 0xffffffff) { z := shr(32, z) r := add(r, 32) }
            if gt(z, 0xffff) { z := shr(16, z) r := add(r, 16) }
            if gt(z, 0xff) { r := add(r, 8) }
            idx := sub(31, shr(3, r))
        }
    }

    function _tailMask(uint256 n) internal pure returns (uint256 m) {
        assembly ("memory-safe") {
            m := not(shr(mul(n, 8), not(0))) // n >= 32 -> all ones
        }
    }

    /// @dev Positions of the first two '.' separators, 32 bytes per step. A third '.' is rejected later
    ///      because '.' is outside the base64/base64url alphabet, so the signature segment needs no scan.
    function _dots(bytes calldata jwt) internal pure returns (uint256 d1, uint256 d2) {
        uint256 len = jwt.length;
        uint256 pat = 0x2e * ONES;
        uint256 found;
        uint256 off;
        assembly ("memory-safe") {
            off := jwt.offset
        }
        for (uint256 p; p < len && found < 2;) {
            uint256 w;
            assembly ("memory-safe") {
                w := calldataload(add(off, p))
            }
            uint256 z = _zeroMask(w ^ pat) & _tailMask(len - p);
            if (z == 0) {
                p += 32;
                continue;
            }
            uint256 q = p + _firstFlag(z);
            if (found == 0) d1 = q;
            else d2 = q;
            ++found;
            p = q + 1;
        }
        if (found != 2 || d1 == 0 || d2 == d1 + 1 || d2 + 1 == len) revert MalformedJwt();
    }

    function _at(bytes memory j, uint256 i) internal pure returns (uint256 c) {
        if (i >= j.length) revert MalformedJson(i);
        assembly ("memory-safe") {
            c := byte(0, mload(add(add(j, 0x20), i)))
        }
    }

    function _hashRange(bytes memory j, uint256 a, uint256 b) internal pure returns (bytes32 h) {
        assembly ("memory-safe") {
            h := keccak256(add(add(j, 0x20), a), sub(b, a))
        }
    }

    function _strHash(bytes memory j, Span memory s, bytes32 key) internal pure returns (bytes32) {
        if (s.kind == 0) revert MissingClaim(key);
        if (s.kind != T_STRING) revert WrongType(key);
        if (s.esc) revert EscapedClaim(key); // no escape decoding: fail closed
        return _hashRange(j, s.start, s.end);
    }

    function _uint(bytes memory j, Span memory s, bytes32 key) internal pure returns (uint256 r) {
        if (s.kind == 0) revert MissingClaim(key);
        if (s.kind != T_NUMBER || s.end - s.start > 15) revert WrongType(key);
        for (uint256 i = s.start; i < s.end; ++i) {
            uint256 ch = _at(j, i);
            if (ch < 0x30 || ch > 0x39) revert WrongType(key); // rejects '-', '.', 'e', '+'
            r = r * 10 + (ch - 0x30);
        }
    }

    /// @dev amr must be an array of plain strings, contain "pop" and NOT contain "mfa".
    ///      World: ["pop"] = World ID proof, ["pop","face","user"] = proof + face check,
    ///      ["pop","user","mfa"] = passkey assertion WITHOUT a fresh World proof (so "pop" alone is not enough).
    function _checkAmr(bytes memory j, Span memory s) internal pure {
        if (s.kind == 0) revert MissingClaim(K_AMR);
        if (s.kind != T_ARRAY) revert WrongType(K_AMR);
        uint256 i = _ws(j, s.start + 1); // after '['
        bool pop;
        if (_at(j, i) != 0x5d) {
            while (true) {
                if (_at(j, i) != 0x22) revert WrongType(K_AMR);
                (uint256 b,) = _skipString(j, i);
                bytes32 h = _hashRange(j, i + 1, b - 1);
                if (h == POP) pop = true;
                else if (h == MFA) revert PasskeyOnly();
                i = _ws(j, b);
                uint256 c = _at(j, i);
                if (c == 0x2c) i = _ws(j, i + 1);
                else if (c == 0x5d) break;
                else revert MalformedJson(i);
            }
        }
        if (!pop) revert NoProofOfPersonhood();
    }

    /// @dev Scan a single top-level JSON object. For each key in `keys` returns the span of its value.
    function _scan(bytes memory j, bytes32[] memory keys) internal pure returns (Span[] memory out) {
        out = new Span[](keys.length);
        uint256 len = j.length;
        uint256 i = _ws(j, 0);
        if (_at(j, i) != 0x7b) revert MalformedJson(i); // '{'
        i = _ws(j, i + 1);
        if (_at(j, i) == 0x7d) {
            if (_ws(j, i + 1) != len) revert MalformedJson(i);
            return out;
        }
        while (true) {
            // key
            if (_at(j, i) != 0x22) revert MalformedJson(i);
            (uint256 ke, bool kesc) = _skipString(j, i);
            if (kesc) revert EscapedKey(i); // "nonce" must not alias "nonce"
            bytes32 kh = _hashRange(j, i + 1, ke - 1);
            i = _ws(j, ke);
            if (_at(j, i) != 0x3a) revert MalformedJson(i); // ':'
            i = _ws(j, i + 1);
            // value
            uint256 vs = i;
            uint256 c = _at(j, i);
            uint8 kind;
            bool vesc;
            if (c == 0x22) {
                kind = T_STRING;
                (i, vesc) = _skipString(j, i);
            } else if (c == 0x7b || c == 0x5b) {
                kind = c == 0x5b ? T_ARRAY : T_OTHER;
                i = _skipContainer(j, i);
            } else {
                // number / true / false / null: run up to a delimiter
                kind = (c == 0x2d || (c >= 0x30 && c <= 0x39)) ? T_NUMBER : T_OTHER;
                while (i < len) {
                    c = _at(j, i);
                    if (c == 0x2c || c == 0x7d || c == 0x20 || c == 0x0a || c == 0x0d || c == 0x09) break;
                    if (c == 0x22 || c == 0x7b || c == 0x5b || c == 0x5d || c == 0x3a) revert MalformedJson(i);
                    ++i;
                }
            }
            // record if wanted
            for (uint256 k; k < keys.length; ++k) {
                if (keys[k] == kh) {
                    if (out[k].kind != 0) revert DuplicateClaim(kh);
                    out[k] = kind == T_STRING ? Span(vs + 1, i - 1, kind, vesc) : Span(vs, i, kind, false);
                    break;
                }
            }
            i = _ws(j, i);
            c = _at(j, i);
            if (c == 0x2c) {
                i = _ws(j, i + 1);
                continue;
            }
            if (c != 0x7d) revert MalformedJson(i);
            if (_ws(j, i + 1) != len) revert MalformedJson(i); // nothing after the object
            return out;
        }
    }

    /// @dev `i` points at an opening quote; returns the index just after the closing quote and whether the
    ///      string contained a backslash escape. Finds the next '"' or '\\' 32 bytes per step (SWAR).
    ///      Raw control characters inside strings are NOT rejected (the signer is trusted to emit valid JSON;
    ///      they cannot change where a string ends).
    function _skipString(bytes memory j, uint256 i) internal pure returns (uint256 end, bool esc) {
        uint256 len = j.length;
        uint256 p = i + 1;
        while (p < len) {
            uint256 w;
            assembly ("memory-safe") {
                w := mload(add(add(j, 0x20), p))
            }
            uint256 z = (_zeroMask(w ^ (0x22 * ONES)) | _zeroMask(w ^ (0x5c * ONES))) & _tailMask(len - p);
            if (z == 0) {
                p += 32;
                continue;
            }
            uint256 k = _firstFlag(z);
            if (uint8(w >> (8 * (31 - k))) == 0x22) return (p + k + 1, esc);
            esc = true; // backslash: skip it and the escaped byte
            p = p + k + 2;
        }
        revert MalformedJson(p);
    }

    /// @dev `i` points at '{' or '['; returns the index just after the closer that brings depth back to 0.
    ///      Strings are skipped with escape tracking, so brackets/quotes inside string values are ignored.
    ///      Bracket TYPES are not matched (assumes the signer emits valid JSON; see header comment).
    function _skipContainer(bytes memory j, uint256 i) internal pure returns (uint256) {
        uint256 depth;
        while (true) {
            uint256 c = _at(j, i);
            if (c == 0x22) {
                (i,) = _skipString(j, i);
                continue;
            }
            if (c == 0x7b || c == 0x5b) ++depth;
            else if (c == 0x7d || c == 0x5d) --depth;
            ++i;
            if (depth == 0) return i;
        }
        revert(); // unreachable
    }

    function _ws(bytes memory j, uint256 i) internal pure returns (uint256 r) {
        assembly ("memory-safe") {
            let base := add(j, 0x20)
            let len := mload(j)
            r := i
            for {} lt(r, len) { r := add(r, 1) } {
                let c := byte(0, mload(add(base, r)))
                if iszero(or(or(eq(c, 0x20), eq(c, 0x0a)), or(eq(c, 0x0d), eq(c, 0x09)))) { break }
            }
        }
    }
}

pragma solidity ^0.8.24;

import {HumanApprovalVerifier as V} from "./HumanApprovalVerifier.sol";

/// @notice Run as an eth_call CREATE against a live Arc RPC (nothing is broadcast):
///         deploys the verifier in the call's scratch state, registers the key, verifies a token and returns
///         abi.encode(modexpOk, modexpGas, modexpOutHash, verifyGas, consumeGas, authTime, chainid, timestamp).
contract ArcProbe {
    struct In {
        bytes jwt;
        bytes n;
        bytes sig;
        bytes32 nonce;
        string sub;
        string aud;
    }

    function _modexp(bytes memory sig, bytes memory n) internal view returns (bool ok, uint256 gasUsed, bytes32 outHash) {
        bytes memory input = abi.encodePacked(uint256(sig.length), uint256(3), uint256(n.length), sig, hex"010001", n);
        uint256 g = gasleft();
        bytes memory out;
        (ok, out) = address(5).staticcall(input);
        gasUsed = g - gasleft();
        outHash = keccak256(out);
    }

    constructor(In memory a) {
        // 1. raw modexp precompile (0x05): sig^65537 mod n
        (bool ok, uint256 modexpGas, bytes32 outHash) = _modexp(a.sig, a.n);
        bytes memory jwt = a.jwt;
        bytes32 nonce = a.nonce;
        string memory sub = a.sub;
        string memory aud = a.aud;
        bytes memory n = a.n;

        // 2. full verifier
        string[] memory acrs = new string[](1);
        acrs[0] = "https://world.org/oidc/acr/orb";
        V v = new V("https://auth.world.org", 3600, acrs);
        v.registerKey(n);
        uint256 g = gasleft();
        v.verify(jwt, nonce, sub, aud);
        uint256 verifyGas = g - gasleft();
        g = gasleft();
        V.Claims memory c = v.verifyAndConsume(jwt, nonce, sub, aud);
        uint256 consumeGas = g - gasleft();

        bytes memory ret = abi.encode(ok, modexpGas, outHash, verifyGas, consumeGas, c.authTime, block.chainid, block.timestamp);
        assembly {
            return(add(ret, 0x20), mload(ret))
        }
    }
}

contract HumanApprovalVerifierTest is Test {
    V v;
    string constant ISS = "https://auth.world.org";

    function setUp() public {
        string[] memory acrs = new string[](2);
        acrs[0] = "https://world.org/oidc/acr/orb";
        acrs[1] = "https://world.org/oidc/acr/orb-face";
        v = new V(ISS, 300, acrs); // World proof must be <= 5 min old
